void cargar_datos(char nombres[5][20],float notas[5]){
    char nombre[20];
    float nota;
    for (int i=0;i<5;i++){
        printf("Ingrese nombre: ");
        scanf(" %s",nombre);
        printf("Ingrese nota: ");
        scanf("%f",&nota);
        strcpy(nombres[i],nombre);
        notas[i]=nota;
    }
}

int maximo(float notas[5]){
    int max_i=0;
    for (int i=1;i<5;i++){
        if (notas[i]>notas[max_i]){
            max_i=i;
        }
    }
    return max_i;
}

int main(){
    char nombres[5][20];
    float notas[5];
    cargar_datos(nombres, notas);

    int max_i=maximo(notas);
    printf("La nota mas alta (%f) la tiene %s\n",notas[max_i],nombres[max_i]);

    char abanderado[20];
    strcpy(abanderado, "Lucas");
    if(strcmp(nombres[max_i],abanderado)==0){
        printf("El abanderado es Lucas");
    }
    else{
        printf("El abanderado no es Lucas");
    }
    return 0;
}

letra=getchar() --scanf de un char
putchar(char) printea una letra
si se ingresan varios a la vez, obtiene/usa solo el primero y lo quita (subisguientes getchars usan los restantes del input) (como si fuera una cola)
//se puede hacer letra==letra/letra!=letra, compara los asciis

ej:
int main(){
    int cont=0;
    char letra;
    printf("Ingrese una oracion, hasta el .: ");
    letra=getchar();
    while(letra!='.'){
        if (letra=='a'){
            cont+=1;
        }
        if (letra!=' '){
            putchar(letra);
        }
        letra=getchar();
    }
    printf("\nCantidad de a: %i",cont);
    return 0;
}
