#include <stdio.h>

int sumar(int a, int b){
    int total=a+b;
    return total;
}

int restar(int a, int b);

int multiplicar(int a, int b){
    int total=a*b;
    return total;
}

int dividir(int a, int b){
    int total=a/b;
    return total;
}

int main(){
    char signo;
    int a,b,res;
    printf("Ingrese el primer numero\n");
    scanf("%d",&a);
    printf("Ingrese el segundo numero\n");
    scanf("%d",&b);
    printf("Ingrese la operacion\n");
    scanf(" %c",&signo);
    while (signo!='+' && signo!='-' && signo!='*' && signo!='/'){
        printf("Signo invalido, ingrese la operacion\n");
        scanf(" %c",&signo);
    }
    switch (signo){
        case '+':
            res=sumar(a,b);
            printf("%d+%d=%d\n",a,b,res);
            break;
        case '-':
            res=restar(a,b);
            printf("%d-%d=%d\n",a,b,res);
            break;
        case '*':
            res=multiplicar(a,b);
            printf("%d*%d=%d\n",a,b,res);
            break;
        default:
            if(b!=0){
                res=dividir(a,b);
                printf("%d/%d=%d\n",a,b,res);
            }
            else{
                printf("No se puede dividir por cero");
            }
    }
}

int restar(int a, int b){
    int res=a-b;
    return res;
}
